def retain(path, relative):
    assert path.is_file() and not path.is_symlink(), 'UNSAFE_EVIDENCE_PATH'
    raw = path.read_bytes()
    assert len(raw) <= 2 * 1024 * 1024, 'EVIDENCE_LIMIT'
    raw.decode('utf-8')
    # Keep raw bytes, while preventing discovery/execution of historical code/logs.
    if relative.endswith(('.py', '.log')):
        relative += '.txt'
    name = prefix + '/' + relative
    target = work / name
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists():
        assert target.read_bytes() == raw, 'EXISTING_EVIDENCE_DIFFERS'
    else:
        target.write_bytes(raw)
    names.append(name)
    mapping[name] = {'sha256': sha(raw), 'source': str(path)}
