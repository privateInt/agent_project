class Result(unittest.TextTestResult):
    def __init__(self,*args,**kwargs):
        super().__init__(*args,**kwargs);self.success_ids=[];self.subcases=[]
    def addSuccess(self,test):
        self.success_ids.append(test.id());super().addSuccess(test)
    def addSubTest(self,test,subtest,error):
        self.subcases.append(dict(test=test.id(),subtest=str(subtest),
            status='PASS' if error is None else 'FAIL' if issubclass(error[0],test.failureException) else 'ERROR'))
        super().addSubTest(test,subtest,error)

def execute(suite,directory):
    module.OBSERVATIONS.clear();module.HARNESS_RESULTS.clear()
    started=datetime.now(timezone.utc).isoformat();begin=time.monotonic()
    with (directory/'unittest.log').open('x') as stream:
        result=unittest.TextTestRunner(stream=stream,verbosity=2,resultclass=Result).run(suite)
    data=dict(context,started_utc=started,elapsed_seconds=time.monotonic()-begin,
        methods_run=result.testsRun,passed_methods=result.success_ids,
        failures=len(result.failures),errors=len(result.errors),skipped=len(result.skipped),
        failure_tests=[str(t) for t,_ in result.failures],error_tests=[str(t) for t,_ in result.errors],
        subcases=result.subcases,observations=list(module.OBSERVATIONS),
        fresh_harness_results=list(module.HARNESS_RESULTS),passed=result.wasSuccessful())
    (directory/'result.json').write_text(json.dumps(data,indent=2)+'\n')
    return data
