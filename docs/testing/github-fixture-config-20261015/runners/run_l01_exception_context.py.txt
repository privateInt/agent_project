sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
stage=sys.argv[1]
assert stage in ('before','after')
expected_source={'before':'487dc146b24b302794dc650128f6aca1349e887e28946976148389dfb603e122',
    'after':'63f5faf6a0bb828928d14bc97a173388ffaca8829f7bad4621a6981e8f180e88'}[stage]
assert sha(SOURCE)==expected_source

old_tree,new_tree=ast.parse(old,feature_version=(3,10)),ast.parse(new,feature_version=(3,10))
def methods(tree,raw):
    result={}
    def walk(node,prefix=''):
        for child in ast.iter_child_nodes(node):
            name=prefix
            if isinstance(child,(ast.FunctionDef,ast.ClassDef)):
                name=prefix+child.name+'.'
                if isinstance(child,ast.FunctionDef):result[name]=ast.get_source_segment(raw,child)
            walk(child,name)
    walk(tree)
    return result
previous,current=methods(old_tree,old),methods(new_tree,new)
assert set(current)-set(previous)=={'Harness.caller_exception_context.'}
assert all(current[n]==s for n,s in previous.items() if n!='Harness.run.')
for node in ast.walk(new_tree):
    if isinstance(node,ast.ClassDef) and node.name=='Harness':
        node.body=[n for n in node.body if not(isinstance(n,ast.FunctionDef) and n.name=='caller_exception_context')]
    if isinstance(node,ast.Dict):
        pairs=[(k,v) for k,v in zip(node.keys,node.values)
               if not(isinstance(k,ast.Constant) and k.value=='exception-context')]
        node.keys=[k for k,v in pairs];node.values=[v for k,v in pairs]
    if isinstance(node,ast.List):
        node.elts=[n for n in node.elts if not(isinstance(n,ast.Tuple) and n.elts
                   and isinstance(n.elts[0],ast.Constant) and n.elts[0].value==TARGET)]
assert ast.dump(new_tree,include_attributes=False)==ast.dump(old_tree,include_attributes=False)

    assert before['test_sha256']==test_pin

module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
