def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def save(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=True, allow_nan=False) + "\n")


def need(condition, reason):
    if not condition:
        raise AssertionError(reason)


def remaining():
    seconds = (DEADLINE - dt.datetime.now(dt.timezone.utc)).total_seconds()
    need(seconds > 0, "FIXED_040000Z_LOCAL_DEADLINE_EXPIRED_NO_EXTENSION")
    return seconds


def mutate(raw, stage):
    tree = ast.parse(raw, feature_version=(3, 10))
    selected = ast.parse(EXPRESSIONS[stage], mode="eval").body
    matches = [n for n in ast.walk(tree) if type(n) is type(selected)
               and ast.dump(n, include_attributes=False) == ast.dump(selected, include_attributes=False)]
    need(len(matches) == 1, "EXACTLY_ONE_REVIEWED_STRICT_PREDICATE_REQUIRED")
    node = matches[0]
    need(isinstance(node, ast.Compare) and len(node.ops) == 1, "ONE_COMPLETE_COMPARISON_REQUIRED")
    lines = raw.splitlines(keepends=True)
    start = sum(map(len, lines[:node.lineno-1])) + node.col_offset
    end = sum(map(len, lines[:node.end_lineno-1])) + node.end_col_offset
    changed = raw[:start] + b"True" + raw[end:]

    class ReplaceExact(ast.NodeTransformer):
        def visit(self, item):
            if type(item) is type(node) and getattr(item, "lineno", None) == node.lineno and getattr(item, "col_offset", None) == node.col_offset:
                return ast.Constant(value=True)
            return super().visit(item)

    expected = ReplaceExact().visit(copy.deepcopy(tree))
    need(ast.dump(expected, include_attributes=False) == ast.dump(ast.parse(changed), include_attributes=False), "NON_TARGET_AST_CHANGED")
    return changed, {"source_sha256": sha(raw), "mutated_source_sha256": sha(changed), "target_count": 1,
                    "before_expression": ast.get_source_segment(raw.decode(), node), "after_expression": "True",
                    "line": node.lineno, "other_ast_equal": True, "other_bytes_equal": True}


class BoundaryViolation(BaseException):
    pass


def worker(copy_root, stage_dir, stage, forbidden_root):
    copy_root, stage_dir = Path(copy_root), Path(stage_dir)
    os.chdir(copy_root)
    sys.dont_write_bytecode = True
    sys.path[:] = [str(copy_root), str(copy_root / "vendor")] + [p for p in sys.path if p
        and not os.path.abspath(p).startswith(forbidden_root) and "site-packages" not in p and "dist-packages" not in p]
    active, violations, events = [True], [], []
    std_roots = tuple(os.path.abspath(p) for p in (sys.base_prefix+"/lib", sys.prefix+"/lib", "/usr/lib", "/lib"))

    def allowed(path, write=False):
        if isinstance(path, int): return True
        if not isinstance(path, (str, bytes, os.PathLike)): return False
        text = os.path.abspath(os.fsdecode(os.fspath(path)))
        if text == str(stage_dir) or text.startswith(str(stage_dir)+os.sep): return True
        if text == str(copy_root) or text.startswith(str(copy_root)+os.sep): return not write
        if text == forbidden_root or text.startswith(forbidden_root+os.sep) or text == "/opt" or text.startswith("/opt/"): return False
        return not write and any(text == root or text.startswith(root+os.sep) for root in std_roots)

    def audit(event, args):
        if not active[0]: return
        denied = event.startswith(("socket.", "subprocess.", "os.exec", "os.spawn", "os.posix_spawn")) or event in (
            "sqlite3.connect", "os.system", "os.fork", "os.forkpty", "os.putenv", "os.unsetenv")
        if event == "open":
            mode, flags = args[1:3]
            write = (isinstance(mode, str) and any(c in mode for c in "wax+")) or (
                isinstance(flags, int) and bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)))
            denied = not allowed(args[0], write)
        elif event in ("os.listdir", "os.scandir"):
            denied = not allowed(args[0] if args else ".")
        elif event in ("os.remove", "os.rmdir", "os.mkdir", "os.chmod", "os.truncate", "os.rename", "os.link", "os.symlink"):
            denied = True
        if denied:
            violations.append({"event": event, "reason": "FAKE_ONLY_NO_EXTERNAL_IO_BOUNDARY"})
            raise BoundaryViolation("NATIVE_OR_ORIGINAL_PRIVATE_SOURCE_IO_FORBIDDEN")
        if event == "open": events.append("ALLOWED_COPY_OR_STDLIB_OR_EVIDENCE_READ")

    sys.addaudithook(audit)
    original_fs = {}
    ancestors = {str(p) for base in (copy_root, *(Path(p) for p in std_roots)) for p in base.parents}
    for name in ("stat", "lstat", "listdir", "scandir", "access", "readlink"):
        original_fs[name] = getattr(os, name)

        def guarded(path=".", *args, _actual=original_fs[name], _kind=name, **kwargs):
            ancestor = (_kind in ("stat", "lstat", "access", "readlink") and isinstance(path, (str, bytes, os.PathLike))
                        and os.path.abspath(os.fsdecode(os.fspath(path))) in ancestors)
            if active[0] and not allowed(path) and not ancestor:
                violations.append({"event": "filesystem_metadata", "reason": "PRIVATE_PATH_FORBIDDEN"})
                raise BoundaryViolation("ORIGINAL_OPT_OPERATOR_METADATA_FORBIDDEN")
            return _actual(path, *args, **kwargs)
        setattr(os, name, guarded)

    def expired(signum, frame):
        raise BoundaryViolation("FIXED_LOCAL_DEADLINE_EXPIRED")

    signal.signal(signal.SIGALRM, expired)
    signal.setitimer(signal.ITIMER_REAL, remaining())
    data = {}
    try:
        module = importlib.import_module(MODULE)
        need(Path(module.__file__).is_relative_to(copy_root) and module.WORK == copy_root, "PORTABLE_MINIMAL_SOURCE_REQUIRED")
        if stage == "normal":
            cases = [module.ReadOnlyMetadataTests(name) for name in METHODS]
            suite = unittest.TestSuite(cases)
        else:
            module.ReadOnlyMetadataTests.setUpClass()
            instance = module.ReadOnlyMetadataTests(METHODS[1]); instance.setUp()
            cases = [instance]
            suite = unittest.TestSuite([unittest.FunctionTestCase(getattr(instance, TARGETS[stage]), description=stage+"."+TARGETS[stage])])
        records = []

        class RecordedResult(unittest.TextTestResult):
            def addSuccess(self, test):
                records.append({"id": test.id(), "outcome": "PASS"}); super().addSuccess(test)

            def addFailure(self, test, err):
                frame, tb = None, err[2]
                while tb:
                    if tb.tb_frame.f_code.co_filename == str(copy_root / TEST):
                        frame = {"function": tb.tb_frame.f_code.co_name, "line": tb.tb_lineno}
                    tb = tb.tb_next
                records.append({"id": test.id(), "outcome": "FAIL", "exception_type": err[0].__name__, "last_test_assertion": frame})
                super().addFailure(test, err)

            def addError(self, test, err):
                records.append({"id": test.id(), "outcome": "ERROR", "exception_type": err[0].__name__}); super().addError(test, err)

            def addSubTest(self, test, subtest, err):
                if err is not None: records.append({"id": test.id(), "outcome": "FAIL" if issubclass(err[0], AssertionError) else "ERROR", "subtest": True})
                super().addSubTest(test, subtest, err)

            def addSkip(self, test, reason):
                records.append({"id": test.id(), "outcome": "SKIP"}); super().addSkip(test, reason); self.stop()

        with (stage_dir / "first.log").open("x") as log:
            result = unittest.TextTestRunner(stream=log, verbosity=2, failfast=True, resultclass=RecordedResult).run(suite)
        data = {"stage": stage, "testsRun": result.testsRun, "failures": len(result.failures), "errors": len(result.errors),
                "skips": len(result.skipped), "methods": records, "boundary_violations": violations, "source_copy": str(copy_root),
                "observations": module.OBSERVATIONS, "fixture_cleanup": module.CLEANUPS, "native_calls": 0, "sqlite_connections": 0, "old_testcase_calls": 0,
                "old_testcase_absent_from_fixture": True, "websocket_version": module.PRIMITIVES.websocket.__version__,
                "websocket_file": module.PRIMITIVES.websocket.__file__, "allowed_reads": len(events)}
        need(Path(module.PRIMITIVES.websocket.__file__).is_relative_to(copy_root / "vendor/websocket"), "PINNED_DEPENDENCY_COPY_REQUIRED")
        need(not any(isinstance(value, type) and issubclass(value, unittest.TestCase) for value in module.PRIMITIVES.__dict__.values()),
             "OLD_TESTCASE_CLASS_MUST_BE_ABSENT")
        if stage == "normal":
            markers = cases[1].helper_markers
            marker_pass = set(markers) == set(TARGETS.values()) and all(v == {"outcome": "PASS", "invocations": 1, "request_count": 0} for v in markers.values())
            data["normal_helper_markers"] = markers
            data["all_pass"] = bool(result.testsRun == 4 and result.wasSuccessful() and not result.skipped and len(records) == 4 and marker_pass and not violations
                and module.CLEANUPS and all(item["all_closed"] and item["forbidden"] == 0 and item["graceful"] == 0 for item in module.CLEANUPS))
        else:
            instance = cases[0]; witness = instance.last_wrong_wire or {}
            raw = base64.b64decode(witness["request_wire_base64"], validate=True) if witness.get("request_wire_base64") else None
            decoded = module.PRIMITIVES.outgoing_frame(raw) if raw else None
            request = witness.get("actual_request")
            wrong = (type(raw) is bytes and sha(raw) == witness.get("request_wire_sha256") and type(request) is dict
                and decoded == (1, request) and witness.get("request_count") == 1 and witness.get("observed_code") is None
                and type(request.get("id")) is int and request["id"] == 1 and witness.get("next_id") == 2
                and request.get("method") == witness.get("expected_method") and request.get("params") == witness.get("expected_params")
                and witness.get("connects") == 1 and witness.get("native_calls") == 0)
            if stage == "M_INCLUDE_TURNS":
                wrong = wrong and request["method"] == "thread/read" and request["params"].get("includeTurns") is True
            else:
                wrong = wrong and request["method"] == "thread/turns/list" and type(request["params"].get("limit")) is int and request["params"]["limit"] == 2
            expression = ast.parse('self.assertEqual("RPC_PARAMS_FORBIDDEN", observed_code)').body[0].value
            assertions = [n for n in ast.walk(ast.parse((copy_root / TEST).read_bytes())) if isinstance(n, ast.Call)
                and ast.dump(n, include_attributes=False) == ast.dump(expression, include_attributes=False)]
            need(len(assertions) == 1, "EXACT_DESIGNATED_SEMANTIC_ASSERTION_REQUIRED")
            expected_frame = {"function": "_blocked_mutation_target", "line": assertions[0].lineno}
            semantic = len(records) == 1 and records[0].get("exception_type") == "AssertionError" and records[0].get("last_test_assertion") == expected_frame
            data["wrong_wire_witness"] = witness
            data["actual_wrong_wire_verified"] = bool(wrong)
            data["helper_invocations"] = instance.helper_invocations
            data["semantic_fail1_error0"] = bool(result.testsRun == 1 and len(result.failures) == 1 and not result.errors
                and not result.skipped and semantic and wrong and instance.helper_invocations == {TARGETS[stage]: 1} and not violations)
    except BaseException as error:
        active[0] = False
        with (stage_dir / "first-harness-error.log").open("x") as log: traceback.print_exc(file=log)
        data = {"stage": stage, "all_pass": False, "harness_error": type(error).__name__, "boundary_violations": violations, "native_calls": 0}
    finally:
        active[0] = False
        signal.setitimer(signal.ITIMER_REAL, 0)
        for name, actual in original_fs.items(): setattr(os, name, actual)
        save(stage_dir / "result.json", data)
    return 0 if data.get("all_pass") or data.get("semantic_fail1_error0") else 1


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--source-ready", action="store_true")
    parser.add_argument("--source-sha256")
    parser.add_argument("--internal-stage", choices=("normal", "M_INCLUDE_TURNS", "M_LIMIT"))
    parser.add_argument("--copy-root"); parser.add_argument("--stage-dir"); parser.add_argument("--forbidden-root")
    args = parser.parse_args()
    if args.internal_stage: return worker(args.copy_root, args.stage_dir, args.internal_stage, args.forbidden_root)
    need(args.source_ready and args.source_sha256 == SOURCE_PIN, "EXPLICIT_SOURCE_READY_EXACT_PIN_REQUIRED")
    remaining()
    raw_ready = (HERE / "tests-ready.json").read_bytes(); ready = json.loads(raw_ready)
    need(sha(Path(__file__).read_bytes()) == ready["runner_sha256"], "RUNNER_PIN_MISMATCH")
    need(ready["source_sha256"] == SOURCE_PIN and ready["original_sha256"] == ORIGINAL_PIN, "READINESS_SOURCE_BASELINE_MISMATCH")
    for name, pin in ready["evidence_pins"].items():
        need(sha((HERE / name).read_bytes()) == pin, "PLAN_REVIEW_ENVIRONMENT_GUARD_PIN_MISMATCH:"+name)
    need(sha((HERE / "plan.md").read_bytes()) == PLAN_PIN and sha((HERE / "plan-addendum.md").read_bytes()) == ADDENDUM_PIN,
         "APPROVED_COMBINED_PLAN_REQUIRED")
    source_ready = json.loads((HERE / "source-ready.json").read_bytes())
    need(source_ready["source_sha256"] == SOURCE_PIN and source_ready["original_sha256"] == ORIGINAL_PIN
         and source_ready["plan_sha256"] == PLAN_PIN and source_ready["addendum_sha256"] == ADDENDUM_PIN, "SOURCE_READY_PLAN_PIN_MISMATCH")
    review = (HERE / "source-review.md").read_text()
    need("판정: **PASS**" in review and SOURCE_PIN in review and "REQUEST_CHANGES" not in review, "INDEPENDENT_SOURCE_STATIC_PASS_REQUIRED")
    environment = json.loads((HERE / "authoring-environment.json").read_bytes())
    need(environment["python_version"] == sys.version and environment["executable"] == sys.executable, "EXECUTOR_ENVIRONMENT_PIN_MISMATCH")
    sources = {}
    for name, pin in ready["files"].items():
        raw = (WORK / name).read_bytes(); need(sha(raw) == pin, "SOURCE_TEST_FIXTURE_PIN_MISMATCH:"+name); sources[name] = raw
    dependencies = {}
    for name, pin in ready["dependencies"].items():
        raw = (HERE / "dependency" / name).read_bytes(); need(sha(raw) == pin, "WEBSOCKET_DEPENDENCY_PIN_MISMATCH:"+name); dependencies[name] = raw
    tree = ast.parse(sources[TEST], feature_version=(3, 10))
    need(tuple(n.name for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name.startswith("test_")) == METHODS, "EXACT_NEW_FOUR_METHODS_REQUIRED")
    guard = json.loads((HERE / "guard-snapshot.json").read_bytes())
    need(guard["source_sha256"] == SOURCE_PIN and guard["expressions"] == EXPRESSIONS, "SOURCE_STRICT_GUARDS_REQUIRED")
    mutated, audits = {}, {}
    for stage in TARGETS:
        mutated[stage], audits[stage] = mutate(sources[PRODUCT], stage)
        need(audits[stage]["line"] == guard["locations"][stage]["line"], "EXACT_GUARD_LOCATION_REQUIRED")
    with (HERE / "execution-attempt.json").open("x") as marker:
        json.dump({"started_utc": dt.datetime.now(dt.timezone.utc).isoformat(), "deadline_utc": DEADLINE.isoformat(),
                   "clock_reset": False, "readiness_sha256": sha(raw_ready), "source_sha256": SOURCE_PIN}, marker, indent=2)
    snapshot = HERE / "initial-snapshot"; snapshot.mkdir()
    for name, raw in sources.items():
        target = snapshot / name; target.parent.mkdir(parents=True, exist_ok=True); target.write_bytes(raw)
    for name, raw in dependencies.items():
        target = snapshot / "vendor" / name; target.parent.mkdir(parents=True, exist_ok=True); target.write_bytes(raw)
    for name in ready["evidence_pins"]:
        target = snapshot / "evidence" / name; target.parent.mkdir(parents=True, exist_ok=True); target.write_bytes((HERE/name).read_bytes())
    (snapshot / "run-once.py").write_bytes(Path(__file__).read_bytes()); (snapshot / "tests-ready.json").write_bytes(raw_ready)
    save(HERE / "execution-environment.json", {"python_version": sys.version, "executable": sys.executable,
         "turn_metadata": environment["declared_turn_metadata"], "native_investigation": "CLOSED", "native_calls": 0,
         "deadline_utc": DEADLINE.isoformat(), "clock_reset": False, "scheduler_PAUSE_changed": False})
    save(HERE / "mutation-ast-audits.json", audits)
    total = {"normal": None, "M_INCLUDE_TURNS": None, "M_LIMIT": None, "native_calls": 0, "old_suites_rerun": False,
             "deadline_utc": DEADLINE.isoformat(), "clock_reset": False}
    for stage in ("normal", "M_INCLUDE_TURNS", "M_LIMIT"):
        stage_dir = HERE / stage; stage_dir.mkdir()
        with tempfile.TemporaryDirectory(prefix="native-metadata-portable-%s-" % stage) as directory:
            copied = Path(directory) / "minimal-source"; copied.mkdir()
            for name, raw in sources.items():
                value = mutated[stage] if stage != "normal" and name == PRODUCT else raw
                target = copied/name; target.parent.mkdir(parents=True, exist_ok=True); target.write_bytes(value)
                need(target.read_bytes() == value, "EXACT_SOURCE_COPY_REQUIRED")
            for name, raw in dependencies.items():
                target = copied/"vendor"/name; target.parent.mkdir(parents=True, exist_ok=True); target.write_bytes(raw)
                need(sha(target.read_bytes()) == ready["dependencies"][name], "EXACT_DEPENDENCY_COPY_REQUIRED")
            (copied/"run-once.py").write_bytes(Path(__file__).read_bytes())
            save(stage_dir/"source-copy-pins.json", {name:sha((copied/name).read_bytes()) for name in sources})
            if stage != "normal": (stage_dir/"mutated-source.py.txt").write_bytes(mutated[stage])
            command = [sys.executable, "-I", "-S", "-B", str(copied/"run-once.py"), "--internal-stage", stage,
                       "--copy-root", str(copied), "--stage-dir", str(stage_dir), "--forbidden-root", str(ROOT)]
            save(stage_dir/"command.json", command)
            with (stage_dir/"process.log").open("x") as log:
                completed = subprocess.run(command, cwd=copied, stdout=log, stderr=subprocess.STDOUT, check=False, timeout=remaining())
            need((stage_dir/"result.json").exists(), "FIRST_CHILD_RESULT_MISSING_STOP_NO_RETRY")
            result = json.loads((stage_dir/"result.json").read_bytes())
            result["process_returncode"] = completed.returncode; result["result_sha256"] = sha((stage_dir/"result.json").read_bytes())
            result["first_log_sha256"] = sha((stage_dir/"first.log").read_bytes()) if (stage_dir/"first.log").exists() else None
            total[stage] = result; save(HERE/"result.json", total)
            ok = result.get("all_pass") if stage == "normal" else result.get("semantic_fail1_error0")
            if completed.returncode or not ok:
                print("STOP "+stage+" evidence="+str(stage_dir)); return 1
    total["normal_helper_pass_reused"] = list(TARGETS.values())
    total["all_required_verification_pass"] = True; total["completed_utc"] = dt.datetime.now(dt.timezone.utc).isoformat()
    save(HERE/"result.json", total)
    print("PASS normal4 + exact2 actual wrong-wire semantic FAIL1/ERROR0 mutants; evidence="+str(HERE))
    return 0


if __name__ == "__main__":
    try:
        exit_status = main()
    except BaseException as error:
        if ROOT is not None:
            with (HERE/"first-controller-error.log").open("x") as first: traceback.print_exc(file=first)
            save(HERE/"first-controller-error.json", {"exception_type":type(error).__name__, "deadline_utc":DEADLINE.isoformat(),
                 "clock_reset":False,"automatic_retry":False})
        raise
    raise SystemExit(exit_status)
