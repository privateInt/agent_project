PRIVATE = "PRIVATE_SYNTHETIC_INTENT_PROMPT_NEVER_IN_JOURNAL"
PRIVATE_ERROR = "PRIVATE_SYNTHETIC_SQL_ERROR_NEVER_PUBLIC"
PROJECT_KEYS = ("project_id", "repository", "dev_root", "git_common_dir")
RECORD_KEYS = {"project_id", "batch_id", "task_id", "client_user_message_id",
               "intent_bytes", "intent_sha256", "state", "revision"}
REPORT_KEYS = {"scope", "status", "reason", "database_commit_confirmed", "transaction_closed",
               "new_claim", "remote_io_performed", "remote_mutation_authorized",
               "input_origin_verified", "native_ready"}
CONFIRMED = {"JOURNAL_READY", "INTENT_STORED", "INTENT_REUSED", "INTENT_OBSERVED", "CLAIM_RECORDED"}
BLOCKED = {"ARGUMENTS_INVALID", "CONNECTION_INVALID", "DATABASE_NOT_EMPTY", "SCHEMA_INVALID",
           "INTENT_INVALID", "INTENT_HASH_MISMATCH", "INTENT_NOT_FOUND", "INTENT_CONFLICT",

           "ROW_INVALID"}
OBSERVATIONS = []


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"),
                      ensure_ascii=True, allow_nan=False).encode("utf-8")


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def synthetic():
    fixture = json.loads(FIXTURE_PATH.read_text())
    if fixture["provenance"] != "SYNTHETIC_ONLY_DEDICATED_TEMP_DB_NO_REMOTE_IO":
        raise AssertionError("Only the explicit synthetic fixture is allowed")
    value = fixture["intent"]
    if (sha(canonical(value)) != fixture["canonical_intent_sha256"]
            or value["project_sha256"] != sha(canonical({k: value[k] for k in PROJECT_KEYS}))
            or value["task_sha256"] != sha(canonical(fixture["task_descriptor"]))
            or value["prompt_sha256"] != sha(fixture["prompt_fixture"].encode("utf-8"))
            or value["settings_sha256"] != sha(canonical(fixture["settings_fixture"]))):
        raise AssertionError("Synthetic fixture digest binding is invalid")
    return value


def changed_intent(**updates):
    value = synthetic()
    value.update(updates)
    value["project_sha256"] = sha(canonical({k: value[k] for k in PROJECT_KEYS}))
    return value


def intent_key(value):
    return tuple(value[k] for k in ("project_id", "batch_id", "task_id"))


def connect(path, *, shared=False):
    connection = sqlite3.connect(path, isolation_level=None, timeout=1, check_same_thread=not shared)
    connection.execute("PRAGMA journal_mode=DELETE").close()
    connection.execute("PRAGMA synchronous=FULL").close()
    connection.execute("PRAGMA busy_timeout=1000").close()
    return connection


def transaction_closed(connection):
    try:
        return type(connection) is sqlite3.Connection and not connection.in_transaction
    except sqlite3.ProgrammingError:
        return False


def profile(connection):
    try:
        if type(connection) is not sqlite3.Connection:
            return None
        pragmas = []
        for name in ("synchronous", "journal_mode", "busy_timeout"):
            cursor = connection.execute("PRAGMA " + name)
            try:
                pragmas.append(tuple(cursor.fetchone()))
            finally:
                cursor.close()
        return (connection.isolation_level, connection.row_factory, connection.text_factory, tuple(pragmas))
    except sqlite3.ProgrammingError:
        return None


def rows(connection):
    return connection.execute(
        "SELECT project_id,batch_id,task_id,client_user_message_id,intent_bytes,"
        "intent_sha256,state,revision FROM submission_intents ORDER BY project_id,batch_id,task_id"
    ).fetchall()


def database_snapshot(connection, path):
    objects = connection.execute("SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name").fetchall()
    names = {row[1] for row in objects if row[0] == "table"}
    data = {}
    for name in ("submission_intents", "foreign_sentinel"):
        if name in names:
            data[name] = connection.execute("SELECT * FROM " + name + " ORDER BY 1").fetchall()
    return {"objects": objects, "rows": data,
            "app": connection.execute("PRAGMA application_id").fetchone()[0],
            "version": connection.execute("PRAGMA user_version").fetchone()[0],
            "bytes": path.read_bytes()}


class ForbiddenIO(BaseException):
    """An IO attempt must not be hidden by product's ordinary Exception handling."""


class NoClockDateTime(dt.datetime):
    @classmethod
    def now(cls, *args, **kwargs):
        raise ForbiddenIO("CLOCK_READ_FORBIDDEN")

    @classmethod
    def utcnow(cls, *args, **kwargs):
        raise ForbiddenIO("CLOCK_READ_FORBIDDEN")

    @classmethod
    def today(cls, *args, **kwargs):
        raise ForbiddenIO("CLOCK_READ_FORBIDDEN")


@contextlib.contextmanager
def no_external_io():
    attempts, output = [], io.StringIO()

    def denied(*args, **kwargs):
        attempts.append("FORBIDDEN_EXTERNAL_IO")
        raise ForbiddenIO("JOURNAL_EXTERNAL_IO_FORBIDDEN")

    with contextlib.ExitStack() as stack:
        for module, names in (
            (sqlite3, ("connect",)), (builtins, ("open",)), (io, ("open",)),
            (os, ("open", "stat", "lstat", "listdir", "scandir", "system", "getenv")),
            (Path, ("open", "read_bytes", "read_text", "write_bytes", "write_text", "exists")),
            (socket, ("socket", "create_connection", "socketpair", "getaddrinfo")),
            (subprocess, ("Popen", "run", "check_output")),
            (time, ("time", "monotonic", "perf_counter", "process_time", "clock_gettime")),
        ):
            for name in names:
                stack.enter_context(mock.patch.object(module, name, denied))
        stack.enter_context(mock.patch.object(dt, "datetime", NoClockDateTime))
        stack.enter_context(contextlib.redirect_stdout(output))
        stack.enter_context(contextlib.redirect_stderr(output))
        stack.enter_context(mock.patch.object(builtins, "__import__", denied))
        yield attempts, output


class IntentJournalTests(unittest.TestCase):
    @contextlib.contextmanager
    def database(self, *, initialized=True, shared=False):
        with tempfile.TemporaryDirectory(prefix="u01b1-intent-test-") as directory:
            path = Path(directory) / "intent.sqlite"
            connection = connect(path, shared=shared)
            try:
                if initialized:
                    self.invoke("initialize_intent_journal", connection, reason="JOURNAL_READY")
                yield connection, path
            finally:
                connection.close()

    def assert_result(self, result, connection, reason=None):
        self.last_result = result
        self.assertIs(type(result), dict)
        self.assertEqual({"report", "record"}, set(result))
        report, record = result["report"], result["record"]
        self.assertIs(type(report), dict)
        self.assertEqual(REPORT_KEYS, set(report))
        self.assertEqual("PRE_SUBMISSION_JOURNAL_ONLY", report["scope"])
        self.assertIn(report["reason"], CONFIRMED | BLOCKED | {"CHECKPOINT_UNCONFIRMED"})
        confirmed = report["reason"] in CONFIRMED
        expected_status = "CONFIRMED" if confirmed else (
            "UNCERTAIN" if report["reason"] == "CHECKPOINT_UNCONFIRMED" else "BLOCKED")
        self.assertEqual(expected_status, report["status"])
        for key in REPORT_KEYS - {"scope", "status", "reason"}:
            self.assertIs(type(report[key]), bool, key)
        self.assertIs(confirmed, report["database_commit_confirmed"])
        self.assertIs(report["reason"] == "CLAIM_RECORDED", report["new_claim"])
        self.assertEqual(transaction_closed(connection), report["transaction_closed"])
        for key in ("remote_io_performed", "remote_mutation_authorized", "input_origin_verified", "native_ready"):
            self.assertIs(False, report[key])
        public = json.dumps(report, sort_keys=True)
        for marker in (PRIVATE, PRIVATE_ERROR, "fixture-client-T1", "/fixture/intent-clone"):
            self.assertNotIn(marker, public)
        if not confirmed or report["reason"] == "JOURNAL_READY":
            self.assertIsNone(record)
        else:
            self.assertIs(type(record), dict)
            self.assertEqual(RECORD_KEYS, set(record))
            self.assertIs(type(record["intent_bytes"]), bytes)
            self.assertLessEqual(len(record["intent_bytes"]), 16384)
            self.assertEqual(sha(record["intent_bytes"]), record["intent_sha256"])
            value = json.loads(record["intent_bytes"])
            self.assertEqual(canonical(value), record["intent_bytes"])
            self.assertEqual(intent_key(value), tuple(record[k] for k in ("project_id", "batch_id", "task_id")))
            self.assertEqual(value["clientUserMessageId"], record["client_user_message_id"])
            self.assertIs(type(record["revision"]), int)
            self.assertTrue((record["state"] == "PREPARED" and record["revision"] == 0)
                            or (record["state"] == "SUBMISSION_UNKNOWN" and record["revision"] >= 1))
            self.assertNotIn(PRIVATE.encode(), record["intent_bytes"])
            self.assertNotIn(b'"prompt_fixture"', record["intent_bytes"])
            self.assertNotIn(b'"settings_fixture"', record["intent_bytes"])
        OBSERVATIONS.append({"test": self._testMethodName, "case": str(self._subtest),
                             "report": dict(report), "record_revision": None if record is None else record["revision"]})
        if reason is not None:
            self.assertEqual(reason, report["reason"])
        return result

    def invoke(self, name, connection, *args, reason=None):
        original_profile = profile(connection)
        with no_external_io() as (attempts, output):
            result = getattr(product, name)(connection, *args)
        self.assertEqual([], attempts)
        self.assertEqual("", output.getvalue())
        self.assertEqual(original_profile, profile(connection))
        if original_profile is not None:
            # A borrowed connection remains usable even after a blocked call.
            connection.execute("SELECT 1").close()
        return self.assert_result(result, connection, reason)

    def reserve(self, connection, value=None):
        value = synthetic() if value is None else value
        raw = canonical(value)
        result = self.invoke("reserve_intent", connection, sha(raw), raw, reason="INTENT_STORED")
        self.assertEqual(raw, result["record"]["intent_bytes"])
        return value, raw

    def observed(self, connection, value, raw, state, revision):
        result = self.invoke("read_intent", connection, intent_key(value), sha(raw), reason="INTENT_OBSERVED")
        self.assertEqual(raw, result["record"]["intent_bytes"])
        self.assertEqual(sha(raw), result["record"]["intent_sha256"])
        self.assertEqual(value["clientUserMessageId"], result["record"]["client_user_message_id"])
        self.assertEqual((state, revision), (result["record"]["state"], result["record"]["revision"]))
        return result

    def test_empty_initialize_and_borrowed_connection(self):
        value, raw = synthetic(), canonical(synthetic())
        with self.database(initialized=False) as (connection, path):
            empty = database_snapshot(connection, path)
            for name, args in (("reserve_intent", (sha(raw), raw)),
                               ("read_intent", (intent_key(value), sha(raw))),
                               ("claim_intent", (intent_key(value), sha(raw), 0))):
                with self.subTest(no_implicit_init=name):
                    self.invoke(name, connection, *args, reason="SCHEMA_INVALID")
                    self.assertEqual(empty, database_snapshot(connection, path))
            self.invoke("initialize_intent_journal", connection, reason="JOURNAL_READY")
            self.assertEqual(0x41494A31, connection.execute("PRAGMA application_id").fetchone()[0])
            self.assertEqual(1, connection.execute("PRAGMA user_version").fetchone()[0])
            columns = connection.execute("PRAGMA table_info(submission_intents)").fetchall()
            self.assertEqual(["project_id", "batch_id", "task_id", "client_user_message_id",
                              "intent_bytes", "intent_sha256", "state", "revision"],
                             [column[1] for column in columns])
            self.assertEqual([1, 2, 3, 0, 0, 0, 0, 0], [column[5] for column in columns])
            ddl = connection.execute("SELECT sql FROM sqlite_master WHERE name='submission_intents'").fetchone()[0]
            self.assertTrue(ddl.endswith("WITHOUT ROWID"))
            unique_client_indexes = []
            for index in connection.execute("PRAGMA index_list(submission_intents)").fetchall():
                index_name = index[1].replace('"', '""')
                fields = connection.execute('PRAGMA index_info("' + index_name + '")').fetchall()
                if index[2] == 1 and [field[2] for field in fields] == ["client_user_message_id"]:
                    unique_client_indexes.append(index[1])
            self.assertEqual(1, len(unique_client_indexes))
            ready = database_snapshot(connection, path)
            self.invoke("initialize_intent_journal", connection, reason="DATABASE_NOT_EMPTY")
            self.assertEqual(ready, database_snapshot(connection, path))
        for setup in ("CREATE TABLE foreign_sentinel (value TEXT)",
                      "PRAGMA application_id=7", "PRAGMA user_version=7"):
            with self.subTest(foreign_database=setup), self.database(initialized=False) as (connection, path):
                connection.execute(setup).close()
                if setup.startswith("CREATE"):
                    connection.execute("INSERT INTO foreign_sentinel VALUES ('keep-original')").close()
                original = database_snapshot(connection, path)
                self.invoke("initialize_intent_journal", connection, reason="DATABASE_NOT_EMPTY")
                self.assertEqual(original, database_snapshot(connection, path))
        for field, changed in (("row_factory", sqlite3.Row), ("text_factory", bytes), ("isolation_level", "DEFERRED")):
            with self.subTest(connection_attribute=field), self.database(initialized=False) as (connection, _):
                setattr(connection, field, changed)
                self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
        for setting in ("PRAGMA synchronous=NORMAL", "PRAGMA journal_mode=MEMORY", "PRAGMA busy_timeout=1001"):
            with self.subTest(connection_pragma=setting), self.database(initialized=False) as (connection, _):
                connection.execute(setting).close()
                self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
        for kind in ("attached", "temporary"):
            with self.subTest(connection_database=kind), self.database(initialized=False) as (connection, path):
                if kind == "attached":
                    connection.execute("ATTACH DATABASE ? AS extra", (str(path.parent / "extra.sqlite"),)).close()
                else:
                    connection.execute("CREATE TEMP TABLE temp_sentinel (value TEXT)").close()
                original = database_snapshot(connection, path)
                self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
                self.assertEqual(original, database_snapshot(connection, path))
        with contextlib.closing(connect(":memory:")) as connection:
            self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
        with self.database(initialized=False) as (connection, path):
            connection.execute("CREATE TABLE foreign_sentinel (value INTEGER)").close()
            connection.execute("BEGIN IMMEDIATE").close()
            connection.execute("INSERT INTO foreign_sentinel VALUES (77)").close()
            self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
            self.assertTrue(connection.in_transaction)
            self.assertEqual([(77,)], connection.execute("SELECT * FROM foreign_sentinel").fetchall())
            with contextlib.closing(connect(path)) as observer:
                self.assertEqual([], observer.execute("SELECT * FROM foreign_sentinel").fetchall())
            connection.rollback()
        with self.database(initialized=False) as (connection, _):
            connection.close()
            self.invoke("initialize_intent_journal", connection, reason="CONNECTION_INVALID")
        self.invoke("initialize_intent_journal", None, reason="CONNECTION_INVALID")
        # Static ownership check supplements actual post-call usability.
        tree = ast.parse(Path(product.__file__).read_text())
        forbidden = [ast.unparse(node.func) for node in ast.walk(tree)
                     if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
                     and (node.func.attr == "connect" or
                          node.func.attr == "close" and isinstance(node.func.value, ast.Name)
                          and node.func.value.id in ("connection", "conn"))]
        self.assertEqual([], forbidden)

    def test_immutable_reserve_roundtrip_and_conflicts(self):
        with self.database() as (connection, path):
            value, raw = self.reserve(connection)
            original = rows(connection)
            with contextlib.closing(connect(path)) as observer:
                self.observed(observer, value, raw, "PREPARED", 0)
            result = self.invoke("reserve_intent", connection, sha(raw), raw, reason="INTENT_REUSED")
            self.assertEqual(raw, result["record"]["intent_bytes"])
            self.assertEqual(original, rows(connection))
            for field, update in (("repository", "other-owner/agent_project"),
                                  ("dev_root", "/fixture/other-clone"),
                                  ("git_common_dir", "/fixture/other-common-git"),
                                  ("base_sha", "b" * 40), ("plan_sha256", "c" * 64),
                                  ("task_sha256", "d" * 64), ("prompt_sha256", "e" * 64),
                                  ("settings_sha256", "f" * 64), ("clientUserMessageId", "fixture-client-replacement")):
                with self.subTest(immutable_field=field):
                    other_raw = canonical(changed_intent(**{field: update}))
                    self.invoke("reserve_intent", connection, sha(other_raw), other_raw, reason="INTENT_CONFLICT")
                    self.assertEqual(original, rows(connection))
            other = changed_intent(task_id="T2", task_sha256="2" * 64)
            other_raw = canonical(other)
            self.invoke("reserve_intent", connection, sha(other_raw), other_raw, reason="CORRELATION_CONFLICT")
            self.assertEqual(original, rows(connection))
            self.invoke("reserve_intent", connection, "0" * 64, raw, reason="INTENT_HASH_MISMATCH")
            self.invoke("reserve_intent", connection, "invalid-pin", raw, reason="ARGUMENTS_INVALID")
            for bad in (list(intent_key(value)), ("vms_project", value["batch_id"], value["task_id"]),
                        ("agent_project",), ("agent_project", "", "T1")):
                with self.subTest(invalid_key=bad):
                    self.invoke("read_intent", connection, bad, sha(raw), reason="ARGUMENTS_INVALID")
            self.invoke("read_intent", connection, intent_key(value), "0" * 64, reason="INTENT_PIN_MISMATCH")
            self.invoke("read_intent", connection, intent_key(other), sha(other_raw), reason="INTENT_NOT_FOUND")
            for revision in (True, -1, 2 ** 63 - 1):
                with self.subTest(invalid_revision=revision):
                    self.invoke("claim_intent", connection, intent_key(value), sha(raw), revision, reason="ARGUMENTS_INVALID")
            invalid = [b"\xef\xbb\xbf" + raw, raw + b"\n", b'{"schema":1,' + raw[1:]]
            for update in ({"schema": True}, {"project_id": "vms_project"},
                           {"plan_sha256": "A" * 64}, {"project_sha256": "0" * 64},
                           {"dev_root": "/fixture/../escape"}, {"git_common_dir": "/fixture/*"},
                           {"task_id": "bad id"}, {"raw_prompt": PRIVATE}):
                bad = copy.deepcopy(value)
                bad.update(update)
                invalid.append(canonical(bad))
            invalid.append(canonical(changed_intent(dev_root="/fixture/" + "r" * 17000)))
            for index, bad_raw in enumerate(invalid):
                with self.subTest(invalid_intent=index):
                    self.invoke("reserve_intent", connection, sha(bad_raw), bad_raw, reason="INTENT_INVALID")
                    self.assertEqual(original, rows(connection))
            connection.close()
            with contextlib.closing(connect(path)) as reopened:
                self.observed(reopened, value, raw, "PREPARED", 0)
                self.assertEqual(original, rows(reopened))
        for field, bad in (("intent_bytes", b"corrupt"), ("intent_sha256", "0" * 64),
                           ("client_user_message_id", "changed-correlation"),
                           ("intent_bytes", b"x" * 16385)):
            with self.subTest(damaged_row=field, length=len(bad)), self.database() as (connection, _):
                value, raw = self.reserve(connection)
                connection.execute("UPDATE submission_intents SET " + field + "=?", (bad,)).close()
                damaged = rows(connection)
                self.invoke("read_intent", connection, intent_key(value), sha(raw), reason="ROW_INVALID")
                self.assertEqual(damaged, rows(connection))

    def test_stale_revision_rejected(self):
        # Single M1 case: only expected revision is wrong; all bindings are valid.
        with self.database() as (connection, _):
            value, raw = self.reserve(connection)

    def test_unknown_never_rearmed(self):
        # Single M2 case: revision1 matches; UNKNOWN is the sole unmet predicate.
        with self.database() as (connection, path):
            value, raw = self.reserve(connection)
            self.invoke("claim_intent", connection, intent_key(value), sha(raw), 0, reason="CLAIM_RECORDED")
            original = rows(connection)
            connection.close()
            with contextlib.closing(connect(path)) as reopened:
                self.invoke("claim_intent", reopened, intent_key(value), sha(raw), 1, reason="UNKNOWN_NO_REPLAY")

                self.observed(reopened, value, raw, "SUBMISSION_UNKNOWN", 1)

    def test_claim_commit_boundary_reopen_and_failures(self):
        with self.database() as (connection, path), contextlib.closing(connect(path)) as observer:
            value, raw = self.reserve(connection)
            actual_commit = product._commit
            boundaries = []

            def commit_boundary(target):
                before = observer.execute("SELECT state,revision FROM submission_intents").fetchone()
                boundaries.append((before, target.in_transaction))
                actual_commit(target)

            with mock.patch.object(product, "_commit", side_effect=commit_boundary):
                self.invoke("claim_intent", connection, intent_key(value), sha(raw), 0, reason="CLAIM_RECORDED")
            self.assertEqual([(("PREPARED", 0), True)], boundaries)
            self.observed(observer, value, raw, "SUBMISSION_UNKNOWN", 1)
            existing = rows(connection)
            self.invoke("reserve_intent", connection, sha(raw), raw, reason="INTENT_REUSED")
            self.assertEqual(existing, rows(connection))
            connection.close()
            with contextlib.closing(connect(path)) as reopened:
                self.observed(reopened, value, raw, "SUBMISSION_UNKNOWN", 1)
        # One shared guard surrounds both workers; avoid racing global mock stacks.
        with self.database(shared=True) as (first, path), contextlib.closing(connect(path, shared=True)) as second:
            value, raw = self.reserve(first)
            ready = threading.Barrier(3)
            results, errors = [], []

            def compete(connection):
                try:
                    ready.wait(timeout=5)
                    results.append((connection, product.claim_intent(connection, intent_key(value), sha(raw), 0)))
                except BaseException as error:
                    errors.append(error)

            workers = [threading.Thread(target=compete, args=(c,), daemon=True) for c in (first, second)]
            with no_external_io() as (attempts, output):
                for worker in workers:
                    worker.start()
                ready.wait(timeout=5)
                for worker in workers:
                    worker.join(timeout=5)
            self.assertTrue(all(not worker.is_alive() for worker in workers))
            self.assertEqual([], errors)
            self.assertEqual([], attempts)
            self.assertEqual("", output.getvalue())
            self.assertEqual(2, len(results))
            self.assertEqual(["CLAIM_RECORDED", "REVISION_CONFLICT"],
                             sorted(result["report"]["reason"] for _, result in results))
            self.assertEqual(1, sum(result["report"]["new_claim"] for _, result in results))
            for connection, result in results:
                self.assert_result(result, connection)
            self.observed(first, value, raw, "SUBMISSION_UNKNOWN", 1)
        stages = ("begin_before", "begin_after", "sql_before", "sql_after", "commit_before",
                  "commit_after", "rollback_before", "rollback_after")
        for stage in stages:
            with self.subTest(failure_stage=stage), self.database() as (connection, path):
                value, raw = self.reserve(connection)
                actual_execute, actual_commit, actual_rollback = product._execute, product._commit, product._rollback
                calls = []

                def execute_hook(target, sql, parameters=()):
                    calls.append(sql)
                    beginning = sql == "BEGIN IMMEDIATE"
                    updating = sql.startswith("UPDATE main.submission_intents ")
                    if (stage == "begin_before" and beginning) or (stage == "sql_before" and updating):
                        raise sqlite3.OperationalError(PRIVATE_ERROR)
                    result = actual_execute(target, sql, parameters)
                    if ((stage == "begin_after" and beginning)
                            or (stage in ("sql_after", "rollback_before", "rollback_after") and updating)):
                        raise sqlite3.OperationalError(PRIVATE_ERROR)
                    return result

                def commit_hook(target):
                    if stage == "commit_before":
                        raise sqlite3.OperationalError(PRIVATE_ERROR)
                    actual_commit(target)
                    if stage == "commit_after":
                        raise sqlite3.OperationalError(PRIVATE_ERROR)

                def rollback_hook(target):
                    if stage == "rollback_before":
                        raise sqlite3.OperationalError(PRIVATE_ERROR)
                    actual_rollback(target)
                    if stage == "rollback_after":
                        raise sqlite3.OperationalError(PRIVATE_ERROR)

                with mock.patch.object(product, "_execute", side_effect=execute_hook), \
                        mock.patch.object(product, "_commit", side_effect=commit_hook), \
