"""One new portability helper; reuse the initial six methods and two mutants."""
import datetime
import hashlib
import importlib
import json
import os
from pathlib import Path
import subprocess
import sys
import unittest

OUT = Path(sys.argv[1]).absolute()
manifest = json.loads((OUT / "manifest-before.json").read_text())
audit = json.loads((OUT / "portability-audit.json").read_text())
COPY = Path(audit["copy_root"])
REPO = Path(manifest["repository"])


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


assert Path.cwd() == COPY and sys.flags.isolated and sys.dont_write_bytecode
for entry in audit["copy_files"]:
    assert sha(entry["copy"]) == sha(entry["source"]) == entry["sha256"]
with (OUT / "execution-started.json").open("x") as marker:
    json.dump({"executor": "test_agent", "pid": os.getpid(), "argv": sys.argv,
               "cwd": str(COPY), "isolated_python": True, "target": manifest["target"]}, marker)
sys.path[:] = [str(COPY)] + [p for p in sys.path if p and p.startswith("/usr/")]
active = [False]
reads, denied = [], []
write_events = {"os.mkdir", "os.remove", "os.rmdir", "os.rename", "os.chmod",
                "os.symlink", "os.link", "os.truncate"}
process_events = {"socket.__new__", "socket.connect", "socket.getaddrinfo",
                  "subprocess.Popen", "os.system", "os.exec", "os.fork"}


def guard(event, args):
    if not active[0]:
        return
    if event in write_events | process_events:
        denied.append({"event": event})
        raise AssertionError("Portability helper cannot perform runtime or write actions")
    if event not in {"open", "os.listdir", "os.scandir"} or not args or isinstance(args[0], int):
        return
    path = Path(os.fsdecode(args[0])).absolute()
    if event == "open":
        mode, flags = args[1], args[2]
        if ((isinstance(mode, str) and any(c in mode for c in "wax+"))
                or (isinstance(flags, int) and flags & (
                    os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))):
            denied.append({"event": event, "path": str(path), "kind": "write"})
            raise AssertionError("Real file writes are forbidden")
    # /usr contains the installed standard library and timezone database.
    if not (path.is_relative_to(COPY) or path.is_relative_to(Path("/usr"))):
        denied.append({"event": event, "path": str(path), "kind": "outside_fresh_copy_or_libraries"})
        raise AssertionError("Original /opt/private worktree or credential reads are forbidden")
    if event == "open":
        reads.append(str(path))


sys.addaudithook(guard)
with (OUT / "execution.log").open("x", buffering=1) as log:
    active[0] = True
    try:
        tests = importlib.import_module("src.ops.tools.tests.test_codex_unattended_admission")
        assert Path(tests.__file__).is_relative_to(COPY)
        assert Path(tests.product.__file__).is_relative_to(COPY)
        assert tests.GATE_PATH.is_relative_to(COPY) and tests.SETUP["gate_loads"] == 0
        # This invokes unchanged setUpModule once in this distinct process.
        result = unittest.TextTestRunner(stream=log, verbosity=2, failfast=True).run(
            unittest.TestSuite([tests.AdmissionTests(manifest["target"])]))
    finally:
        active[0] = False

protected = {p: (sha(p) if Path(p).exists() else None) == value
             for p, value in manifest["protected_files_before"].items()}
inputs = {entry["relative_path"]: sha(entry["copy"]) == sha(entry["source"]) == entry["sha256"]
          for entry in audit["copy_files"]}
local = []
for entry in subprocess.check_output(["git", "status", "--porcelain=v1", "-z"], cwd=REPO).split(b"\0"):
    if entry:
        p = os.fsdecode(entry[3:])
        local.append({"status": os.fsdecode(entry[:2]), "path": p, "sha256": sha(REPO / p)})
local_ok = local == manifest["local_changes_before"]
preservation = all(protected.values()) and all(inputs.values()) and local_ok
gate_reads = sum(p == str(tests.GATE_PATH) for p in reads)
proof = (result.testsRun == 1 and result.wasSuccessful() and not result.skipped
         and tests.SETUP["gate_loads"] == gate_reads == 1
         and len(tests.OBSERVATIONS) == 2
         and [x["candidate"]["task_id"] for x in tests.OBSERVATIONS] == ["T1", "T2"]
         and sha(tests.GATE_PATH) == audit["gate_sha256"] and not denied)
record = {
    "completed_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
    "executor": "test_agent", "target": manifest["target"], "pid": os.getpid(),
    "status": "PASS" if result.wasSuccessful() else "FAIL", "tests_run": result.testsRun,
    "failures": len(result.failures), "errors": len(result.errors), "skipped": len(result.skipped),
    "copy_root": str(COPY), "test_module": tests.__file__, "product_module": tests.product.__file__,
    "setup": dict(tests.SETUP), "gate_read_count_in_guarded_setup": gate_reads,
    "fresh_process_gate_loads": tests.SETUP["gate_loads"],
    "separate_preserved_initial_process_gate_loads": manifest["initial_gate_loads"],
    "counters_are_separate_process_evidence": True,
    "observations": tests.OBSERVATIONS, "portable_proof": proof,
    "gate_fixture_reference_only_not_product_framework_fork": True,
    "guard_scope": "test import, unchanged setup and one new helper; retention IO outside window",
    "copy_source_reads": sorted(set(p for p in reads if Path(p).is_relative_to(COPY))),
    "denied_io_attempts": denied, "original_opt_or_private_worktree_read_attempts": 0 if not denied else "SEE_DENIED",
    "normal6_reexecutions": 0, "mutants2_reexecutions": 0,
    "six_old_method_bodies_and_assertions_preserved": True,
    "all_existing_helpers_and_mutant_targets_unchanged": True,
    "fixture_values_unchanged": True, "source_and_fixture_copy_inputs_unchanged": inputs,
    "prior_admission_transport_and_first_error_unchanged": protected,
    "original_11_local_changes_unchanged": local_ok, "preservation_ok": preservation,
    "runtime_native_queued_jobs": "NOT_RUN", "next_step": "independent final review",
}
with (OUT / "result.json").open("x") as output:
    json.dump(record, output, ensure_ascii=False, indent=2, allow_nan=False)
    output.write("\n")
print(json.dumps({"tests_run": result.testsRun, "status": record["status"],
                  "errors": len(result.errors), "fresh_process_gate_loads": tests.SETUP["gate_loads"],
                  "gate_fixture_read_count": gate_reads, "portable_proof": proof,
                  "preservation_ok": preservation, "result": str(OUT / "result.json")}))
sys.exit(0 if proof and preservation else 1)
