"""Single-use execution record for the four approved offline test methods."""
import datetime
import hashlib
import importlib
import json
import os
from pathlib import Path
import platform
import subprocess
import sys
import traceback
import unittest

OUT = Path(__file__).resolve().parent
manifest = json.loads((OUT / "manifest-before.json").read_text())
WORK = Path(manifest["workdir"])
REPO = WORK.parents[2]
# Exclusive marker: this evidence directory cannot accidentally run twice.
with (OUT / "execution-started.json").open("x") as marker:
    json.dump({"started_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
               "argv": sys.argv, "cwd": str(Path.cwd()), "pid": os.getpid()}, marker)
sys.path.insert(0, str(WORK))
module = importlib.import_module("src.ops.tools.tests.test_codex_daemon_websocket")
names = manifest["method_names"]
actual_names = unittest.TestLoader().getTestCaseNames(module.CodexDaemonWebSocketTests)
assert set(names) == set(actual_names) and len(names) == 4
outcomes = {name: {"status": "RUNNING", "subtests": []} for name in names}


class Result(unittest.TextTestResult):
    def addSuccess(self, test):
        super().addSuccess(test)
        outcomes[test._testMethodName]["status"] = "PASS"

    def addFailure(self, test, err):
        super().addFailure(test, err)
        outcomes[test._testMethodName].update(status="FAIL", error=self._exc_info_to_string(err, test))

    def addError(self, test, err):
        super().addError(test, err)
        outcomes[test._testMethodName].update(status="ERROR", error=self._exc_info_to_string(err, test))

    def addSubTest(self, test, subtest, err):
        super().addSubTest(test, subtest, err)
        row = {"case": str(subtest), "status": "PASS" if err is None else "FAIL"}
        if err:
            row["error"] = self._exc_info_to_string(err, test)
            outcomes[test._testMethodName]["status"] = "FAIL"
        outcomes[test._testMethodName]["subtests"].append(row)


with (OUT / "execution.log").open("x", buffering=1) as log:
    suite = unittest.TestSuite(module.CodexDaemonWebSocketTests(name) for name in names)
    result = unittest.TextTestRunner(stream=log, verbosity=2, resultclass=Result).run(suite)

files_after = {}
for name, entry in manifest["files"].items():
    current = hashlib.sha256(Path(entry["path"]).read_bytes()).hexdigest()
    saved = hashlib.sha256((OUT / name).read_bytes()).hexdigest()
    files_after[name] = {"current_sha256": current, "preserved_sha256": saved,
                         "unchanged": current == saved == entry["sha256"]}
protected_after = {}
for path, before in manifest["protected_results_before"].items():
    p = Path(path)
    after = {"exists": p.exists(), "sha256": hashlib.sha256(p.read_bytes()).hexdigest() if p.exists() else None}
    protected_after[path] = {"unchanged": before == after, **after}
local_after = []
for entry in subprocess.check_output(["git", "status", "--porcelain=v1", "-z"], cwd=REPO).split(b"\0"):
    if entry:
        path = os.fsdecode(entry[3:])
        local_after.append({"status": os.fsdecode(entry[:2]), "path": path,
                            "sha256": hashlib.sha256((REPO / path).read_bytes()).hexdigest()})
preservation_ok = (all(x["unchanged"] for x in files_after.values())
                   and all(x["unchanged"] for x in protected_after.values())
                   and local_after == manifest["local_changes_before"])
record = {
    "completed_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
    "scope": "actual transport/installed decoder, fake socket and business IO only",
    "tests_run": result.testsRun, "method_passes": sum(x["status"] == "PASS" for x in outcomes.values()),
    "failure_records": len(result.failures), "errors": len(result.errors), "skipped": len(result.skipped),
    "successful": result.wasSuccessful(), "outcomes": outcomes, "observations": module.OBSERVATIONS,
    "python": platform.python_version(), "websocket_client": module.websocket.__version__,
    "files_after": files_after, "protected_results_after": protected_after,
    "original_11_local_changes_unchanged": local_after == manifest["local_changes_before"],
    "preservation_ok": preservation_ok,
    "own_current_turn_context": manifest["own_current_turn_context"],
    "real_host_connection": "NOT_RUN", "final_independent_review": "NOT_RUN",
    "legacy_tests": "NOT_RUN", "root_user_pause": "UNCHANGED",
    "permission_probes": 0, "notifications_or_delegation": 0,
}
with (OUT / "result.json").open("x") as target:
    json.dump(record, target, ensure_ascii=False, indent=2)
    target.write("\n")
print(json.dumps({"tests_run": result.testsRun, "method_passes": record["method_passes"],
                  "failure_records": len(result.failures), "errors": len(result.errors),
                  "preservation_ok": preservation_ok, "result": str(OUT / "result.json")}))
sys.exit(0 if result.wasSuccessful() and preservation_ok else 1)
